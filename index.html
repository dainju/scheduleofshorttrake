async function loadCSV() { const response = await fetch("소트 프로그램3.csv"); const text = await response.text(); const rows = text.trim().split("\n"); rows.shift(); // 헤더 제거 const container = document.querySelector(".container"); container.innerHTML = ""; rows.forEach(row => { const cols = row.split(","); const date = cols[0] || ""; const title = cols[1] || ""; const place = cols[2] || ""; const type = cols[3] || ""; const target = cols[4] || ""; const category = cols[5] || ""; container.innerHTML += `
${date}
${title}
${place}

구분: ${type}

참가대상: ${target}

분류: ${category}

`; }); } loadCSV(); function toggleDetail(button){ const detail = button.nextElementSibling; if(detail.style.display==="block"){ detail.style.display="none"; button.innerText="자세히 보기"; }else{ detail.style.display="block"; button.innerText="닫기"; } }
